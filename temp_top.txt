import React, { useState, useEffect } from 'react';
import { motion, AnimatePresence } from 'framer-motion';
import { ArrowRight, AlertCircle, Lock, ChevronLeft, AlertTriangle, ShieldCheck } from 'lucide-react';
import { db, doc, getDoc, updateDoc, arrayUnion } from '../services/firebase';
import { APP_CONFIG } from '../utils/config';
import logo from '../assets/logo.png';

const generateSessionId = () => `${Date.now()}-${Math.random().toString(36).slice(2, 10)}`;
const DEV_PHONE = '8051725780';
const PHONE_MAX = 3;
const PHONE_TIERS = [120, 600, -1];
const PIN_MAX = 5;
const PIN_TIERS = [60, 300, 1800, -1];
const fmtTime = (s) => s >= 60 ? `${Math.floor(s / 60)}m ${s % 60}s` : `${s}s`;

const readPhoneLock = () => {
  try {
    const raw = localStorage.getItem('lg_phone_lock');
    if (!raw) return null;
    const d = JSON.parse(raw);
    if (d.until !== -1 && d.until <= Date.now()) { localStorage.removeItem('lg_phone_lock'); return null; }
    return d;
  } catch { return null; }
};
const writePhoneLock = (until, count, tier) => localStorage.setItem('lg_phone_lock', JSON.stringify({ until, count, tier }));
const clearPhoneLock = () => localStorage.removeItem('lg_phone_lock');
const getRecent = () => { try { return JSON.parse(localStorage.getItem('lg_recent_phones') || '[]'); } catch { return []; } };
const saveRecent = (ph, name) => {
  try {
    const list = getRecent().filter(r => r.phone !== ph);
    list.unshift({ phone: ph, name: name || '' });
    localStorage.setItem('lg_recent_phones', JSON.stringify(list.slice(0, 3)));
  } catch {}
};

export default function Login({ onLogin }) {
  const [step, setStep] = useState('phone');
  const [phone, setPhone] = useState('');
  const [pin, setPin] = useState('');
  const [error, setError] = useState('');
  const [loading, setLoading] = useState(false);
  const [pendingUser, setPendingUser] = useState(null);

  const initLock = readPhoneLock();
  const [phoneLockUntil, setPhoneLockUntil] = useState(initLock?.until ?? 0);
  const [phoneLockTier, setPhoneLockTier] = useState(initLock?.tier ?? 0);
  const [phoneAttempts, setPhoneAttempts] = useState(initLock?.count ?? 0);
  const [phoneCountdown, setPhoneCountdown] = useState(0);
  const [pinAttempts, setPinAttempts] = useState(0);
  const [pinLockTier, setPinLockTier] = useState(0);
  const [pinLockUntil, setPinLockUntil] = useState(0);
  const [pinCountdown, setPinCountdown] = useState(0);
  const [pinBlocked, setPinBlocked] = useState(false);

  useEffect(() => {
    if (step !== 'phone') return;
    const h = (e) => {
      if (e.key >= '0' && e.key <= '9') { setPhone(p => p.length < 10 ? p + e.key : p); setError(''); }
      else if (e.key === 'Backspace') { setPhone(p => p.slice(0, -1)); setError(''); }
      else if (e.key === 'Enter') handlePhoneSubmit();
    };
    window.addEventListener('keydown', h);
    return () => window.removeEventListener('keydown', h);
  }, [step, phone, phoneLockUntil]);

  useEffect(() => {
    if (step !== 'pin') return;
    const h = (e) => {
      if (e.key >= '0' && e.key <= '9') { setPin(p => p.length < 4 ? p + e.key : p); setError(''); }
      else if (e.key === 'Backspace') { setPin(p => p.slice(0, -1)); setError(''); }
      else if (e.key === 'Enter') handlePinSubmit();
    };
    window.addEventListener('keydown', h);
    return () => window.removeEventListener('keydown', h);
  }, [step, pin, pinLockUntil, pinBlocked]);

  useEffect(() => {
    if (phoneLockUntil === -1 || phoneLockUntil <= Date.now()) { setPhoneCountdown(0); return; }
    const tick = () => {
      const r = Math.ceil((phoneLockUntil - Date.now()) / 1000);
      if (r <= 0) { setPhoneCountdown(0); setPhoneLockUntil(0); setPhoneAttempts(0); setError(''); clearPhoneLock(); }
      else setPhoneCountdown(r);
    };
    tick(); const id = setInterval(tick, 1000); return () => clearInterval(id);
  }, [phoneLockUntil]);

  useEffect(() => {
    if (pinBlocked || pinLockUntil <= Date.now()) { setPinCountdown(0); return; }
    const tick = () => {
      const r = Math.ceil((pinLockUntil - Date.now()) / 1000);
      if (r <= 0) { setPinCountdown(0); setPinLockUntil(0); setPinAttempts(0); setError(''); }
      else setPinCountdown(r);
    };
    tick(); const id = setInterval(tick, 1000); return () => clearInterval(id);
  }, [pinLockUntil, pinBlocked]);

  useEffect(() => {
    if (step !== 'pin') return;
    (async () => {
      try {
        const snap = await getDoc(doc(db, 'users', phone));
        if (!snap.exists()) return;
        const d = snap.data();
        if (d.blocked) { setPinBlocked(true); return; }
        if (d.lockUntil) { const until = new Date(d.lockUntil).getTime(); if (until > Date.now()) setPinLockUntil(until); }
        setPinAttempts(d.loginAttempts || 0);
        setPinLockTier(d.lockTier || 0);
      } catch {}
    })();
  }, [step]);

  const handlePhoneSubmit = async () => {
    if (phone.length !== 10) return;
    if (phoneLockUntil === -1) { setError('Device blocked. Contact admin.'); return; }
    if (phoneLockUntil > Date.now()) { setError(`Too many attempts. Wait ${fmtTime(phoneCountdown)}.`); return; }
    setLoading(true); setError('');
    if (phone === DEV_PHONE) { setLoading(false); setPhoneAttempts(0); clearPhoneLock(); setStep('pin'); return; }
    try {
      const snap = await getDoc(doc(db, 'users', phone));
      if (!snap.exists()) {
        const newCount = phoneAttempts + 1;
        if (newCount >= PHONE_MAX) {
          const newTier = phoneLockTier + 1;
          const duration = PHONE_TIERS[Math.min(newTier - 1, PHONE_TIERS.length - 1)];
          if (duration === -1) { setPhoneLockUntil(-1); setPhoneAttempts(newCount); setPhoneLockTier(newTier); writePhoneLock(-1, newCount, newTier); setError('Device blocked. Contact admin.'); }
          else { const until = Date.now() + duration * 1000; setPhoneLockUntil(until); setPhoneAttempts(0); setPhoneLockTier(newTier); writePhoneLock(until, 0, newTier); setError(`Too many attempts. Locked for ${fmtTime(duration)}.`); }
        } else {
          setPhoneAttempts(newCount); writePhoneLock(0, newCount, phoneLockTier);
          const left = PHONE_MAX - newCount;
          setError(`Could not verify. Try again. (${left} attempt${left === 1 ? '' : 's'} left)`);
          setPhone('');
        }
        setLoading(false); return;
      }
      setPhoneAttempts(0); clearPhoneLock(); setStep('pin');
    } catch { setError('Network error. Try again.'); }
    setLoading(false);
  };

  const handlePinSubmit = async () => {
    if (pin.length < 4) return;
    if (pinBlocked) { setError('Account blocked. Contact admin.'); return; }
    if (pinLockUntil > Date.now()) { setError(`Locked. Wait ${fmtTime(pinCountdown)}.`); setPin(''); return; }
    setLoading(true); setError('');
    try {
      if (phone === DEV_PHONE) {
        if (pin !== '0000') { handleWrongPin(); return; }
        await resetPinAttempts(); await createSession({ phone: DEV_PHONE, name: 'Developer', role: 'admin' }); return;
      }
      const snap = await getDoc(doc(db, 'users', phone));
      if (!snap.exists()) { setError('User not found.'); setLoading(false); return; }
      const data = snap.data();
      if (data.blocked && pin !== data.pin) { handleWrongPin(); return; }
      if (data.blocked && pin === data.pin) { await resetPinAttempts(); await createSession({ ...data, phone, blocked: true }); return; }
      if (pin !== data.pin) { handleWrongPin(); return; }
      await resetPinAttempts();
      if (data.role === 'admin') { await createSession({ ...data, phone }); return; }
      if (data.activeSession && data.sessionExpiry && new Date(data.sessionExpiry) > new Date()) {
        setPendingUser({ ...data, phone }); setStep('conflict'); setLoading(false); return;
      }
      await createSession({ ...data, phone });
    } catch { setError('Network error. Try again.'); setLoading(false); }
  };

  const handleWrongPin = async () => {
    const newAttempts = pinAttempts + 1; setPinAttempts(newAttempts); setPin(''); setLoading(false);
    if (newAttempts >= PIN_MAX) {
      const newTier = pinLockTier + 1; setPinLockTier(newTier);
      const duration = PIN_TIERS[Math.min(newTier - 1, PIN_TIERS.length - 1)];
      if (duration === -1) {
        setPinBlocked(true);
        try { await updateDoc(doc(db, 'users', phone), { blocked: true, blockedAt: new Date().toISOString(), blockReason: 'Too many wrong PIN attempts', loginAttempts: 0, lockTier: newTier }); } catch {}
        setError('Account permanently blocked. Contact admin.');
      } else {
        const until = Date.now() + duration * 1000; setPinLockUntil(until); setPinAttempts(0);
        try { await updateDoc(doc(db, 'users', phone), { loginAttempts: 0, lockUntil: new Date(until).toISOString(), lockTier: newTier }); } catch {}
        setError(`Wrong PIN. Locked for ${fmtTime(duration)}.`);
      }
    } else {
      const left = PIN_MAX - newAttempts;
      try { await updateDoc(doc(db, 'users', phone), { loginAttempts: newAttempts, lockTier: pinLockTier }); } catch {}
      setError(`Incorrect PIN ƒ?" ${left} attempt${left === 1 ? '' : 's'} left`);
    }
  };

  const resetPinAttempts = async () => {
    setPinAttempts(0); setPinLockUntil(0); setPinLockTier(0); setPinBlocked(false);
    try { await updateDoc(doc(db, 'users', phone), { loginAttempts: 0, lockUntil: null, lockTier: 0 }); } catch {}
  };

  const createSession = async (userData) => {
    const sessionId = generateSessionId(); const isDev = userData.phone === DEV_PHONE;
    let timeoutHours = 24;
    try { const s = await getDoc(doc(db, 'settings', 'app')); if (s.exists() && s.data().sessionTimeout) timeoutHours = s.data().sessionTimeout; } catch {}
    const sessionExpiry = new Date(Date.now() + (isDev ? 4 : timeoutHours) * 3600000).toISOString();
    saveRecent(userData.phone, userData.name);
    try {
      const userRef = doc(db, 'users', userData.phone); const userSnap = await getDoc(userRef);
      if (userSnap.exists()) {
        const ua = navigator.userAgent || '';
        const upd = { activeSession: sessionId, sessionExpiry, lastLogin: new Date().toISOString(), lastDevice: ua.slice(0, 100) };
        if (userSnap.data().role === 'admin') {
          const device = ua.includes('iPhone') ? 'iPhone' : ua.includes('Android') ? 'Android' : ua.includes('Windows') ? 'Windows PC' : ua.includes('Mac') ? 'Mac' : 'Unknown';
          const browser = ua.includes('Firefox') ? 'Firefox' : ua.includes('Edg') ? 'Edge' : ua.includes('Chrome') ? 'Chrome' : ua.includes('Safari') ? 'Safari' : 'Other';
          const os = ua.includes('Android') ? 'Android' : ua.includes('iPhone') ? 'iOS' : ua.includes('Windows') ? 'Windows' : ua.includes('Mac') ? 'macOS' : ua.includes('Linux') ? 'Linux' : 'Unknown';
          upd.sessions = arrayUnion({ sessionId, device, browser, os, loginAt: new Date().toISOString(), expiresAt: sessionExpiry });
        }
        await updateDoc(userRef, upd);
      }
    } catch {}
    const sessionData = { ...userData, sessionId, sessionExpiry };
    localStorage.setItem(`lg_user_${userData.phone}`, JSON.stringify(sessionData));
    localStorage.setItem('lg_user', JSON.stringify(sessionData));
    localStorage.setItem('lg_last_login', userData.phone);
    sessionStorage.setItem('lg_active_phone', userData.phone);
    onLogin(sessionData);
  };

  const terminateAndLogin = async () => { setLoading(true); await createSession(pendingUser); setLoading(false); };

  // Mobile numpad - centered
  const NumPad = ({ onPress, disabled }) => {
    const t = React.useRef(null);
    return (
      <div className="grid grid-cols-3 gap-3 mt-6 max-w-[280px] mx-auto">
        {[1,2,3,4,5,6,7,8,9,'',0,'ƒO®'].map((k, i) => (
          <motion.button key={i} type="button"
            whileTap={k !== '' ? { scale: 0.84 } : {}}
            transition={{ duration: 0.07 }}
            onClick={() => k !== '' && !disabled && onPress(k)}
            onPointerDown={() => { if (k === 'ƒO®' && !disabled) t.current = setTimeout(() => onPress('CLEAR'), 500); }}
            onPointerUp={() => clearTimeout(t.current)}
            onPointerLeave={() => clearTimeout(t.current)}
            disabled={disabled || k === ''}
            className={`h-14 rounded-2xl font-bold text-xl select-none transition-colors
              ${k === '' ? 'invisible pointer-events-none'
                : k === 'ƒO®' ? 'bg-royal-50 text-royal-600 active:bg-royal-100'
                : 'bg-white text-gray-900 shadow-sm border border-gray-100 active:bg-gray-50'}
              ${disabled ? 'opacity-40 cursor-not-allowed' : ''}`}>
            {k}
          </motion.button>
        ))}
      </div>
    );
  };

  const ErrorLine = ({ show, isLock, text }) => (
    <div className="h-5 flex items-center mt-2">
      <AnimatePresence>
        {show && (
          <motion.p initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}
            className={`text-xs font-medium flex items-center gap-1.5 ${isLock ? 'text-amber-500' : 'text-red-500'}`}>
            {isLock ? <Lock size={10} className="shrink-0" /> : <AlertCircle size={10} className="shrink-0" />}
            {text}
          </motion.p>
        )}
      </AnimatePresence>
    </div>
  );

  const isPhoneLocked = phoneLockUntil === -1 || phoneLockUntil > Date.now();
  const isPinLocked = pinLockUntil > Date.now() || pinBlocked;
  const recent = getRecent();

  return (
    <div className="min-h-screen flex">

      {/* ƒ"?ƒ"? LEFT PANEL - desktop only ƒ"?ƒ"? */}
      <div className="hidden lg:flex lg:w-[44%] xl:w-[40%] bg-[#0f172a] flex-col justify-between p-14 shrink-0 min-h-screen">
        <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} transition={{ delay: 0.1 }}
          className="flex items-center gap-3">
          <img src={logo} alt="Lucy Garden" className="w-11 h-11 rounded-xl object-cover border border-white/10" />
          <div>
            <p className="text-white font-black text-base">Lucy Garden</p>
            <p className="text-white/30 text-xs">Madhubani, Bihar</p>
          </div>
        </motion.div>

        <motion.div initial={{ opacity: 0, y: 16 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.2 }}>
          <h2 className="text-[42px] xl:text-[48px] font-black text-white leading-[1.1] mb-5">
            Simple.<br />Secure.<br />
            <span className="text-royal-400">Reliable.</span>
          </h2>
          <p className="text-white/30 text-sm leading-relaxed max-w-[260px]">
            Sign in to access your account and manage your records.
          </p>
        </motion.div>

        <p className="text-white/15 text-[10px]">
          v{APP_CONFIG.version} Aú Made with ƒ~ & ƒ?‹,? by{' '}
          <a href="https://portfolio-divyanshu-git.vercel.app" target="_blank" rel="noreferrer"
            className="font-bold text-white/30 hover:text-white/50 transition-colors">Divyanshu Gupta</a>
        </p>
      </div>

      {/* ƒ"?ƒ"? RIGHT PANEL ƒ"?ƒ"? */}
      <div className="flex-1 flex flex-col min-h-screen">

        {/* mobile: dark header */}
        <div className="lg:hidden bg-[#0f172a] px-6 pt-14 pb-10 text-center">
          <motion.div initial={{ opacity: 0, y: -10 }} animate={{ opacity: 1, y: 0 }}>
            <img src={logo} alt="Lucy Garden" className="w-14 h-14 rounded-2xl object-cover mx-auto mb-3 border border-white/10 shadow-xl" />
            <p className="text-white font-black text-lg">Lucy Garden</p>
            <p className="text-white/30 text-xs mt-0.5">Madhubani, Bihar</p>
          </motion.div>
        </div>

        {/* form area */}
        <div className="flex-1 bg-[#f8fafc] lg:bg-white flex items-center justify-center px-6 py-8 lg:px-14">
          <div className="w-full max-w-[360px]">
            <motion.div initial={{ opacity: 0, y: 12 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.1 }}>
              <AnimatePresence mode="wait">

                {/* PHONE STEP */}
                {step === 'phone' && (
                  <motion.div key="phone" initial={{ opacity: 0, x: -12 }} animate={{ opacity: 1, x: 0 }} exit={{ opacity: 0, x: -12 }}>
                    <div className="mb-7">
                      <p className="text-[10px] font-bold text-royal-500 tracking-[0.18em] uppercase mb-2 flex items-center gap-1.5">
                        <ShieldCheck size={10} /> Secure Login
                      </p>
                      <p className="text-[10px] font-bold text-royal-500 tracking-[0.18em] uppercase mb-2 flex items-center gap-1.5">
                        <ShieldCheck size={10} /> Secure Login
                      </p>
                      <h2 className="text-2xl font-black text-gray-900">Sign in</h2>
                      <p className="text-sm text-gray-400 mt-1">Enter your registered mobile number</p>
                    </div>

                    {/* recent */}
                    {phone === '' && recent.length > 0 && (
                      <div className="mb-6">
                        <p className="text-[9px] font-bold text-gray-300 uppercase tracking-widest mb-2">Recent</p>
                        <div className="space-y-1.5">
                          {recent.map(r => (
                            <motion.button key={r.phone} type="button" whileTap={{ scale: 0.98 }}
                              onClick={() => { setPhone(r.phone); setError(''); }}
                              className="w-full flex items-center gap-3 px-3 py-2.5 bg-white rounded-xl border border-gray-100 hover:border-royal-200 hover:bg-royal-50/30 transition-all shadow-sm">
                              <div className="w-8 h-8 bg-royal-100 rounded-lg flex items-center justify-center shrink-0">
                                <span className="text-royal-600 font-black text-[11px]">{(r.name || r.phone).slice(0, 2).toUpperCase()}</span>
                              </div>
                              <div className="flex-1 text-left min-w-0">
                                {r.name && <p className="text-sm font-bold text-gray-800 truncate">{r.name}</p>}
                                <p className="text-[11px] font-mono text-gray-400">+91 ƒ?›ƒ?›ƒ?›ƒ?›ƒ?› {r.phone.slice(-5)}</p>
                              </div>
                              <ArrowRight size={12} className="text-gray-300 shrink-0" />
                            </motion.button>
                          ))}
                        </div>
                        <div className="flex items-center gap-3 mt-5">
                          <div className="flex-1 h-px bg-gray-200" />
                          <span className="text-[9px] text-gray-300 uppercase tracking-widest font-bold">or</span>
                          <div className="flex-1 h-px bg-gray-200" />
                        </div>
                      </div>
                    )}

                    {/* phone input */}
                    <motion.div
                      animate={error ? { x: [0, -5, 5, -3, 3, 0] } : {}}
                      transition={{ duration: 0.25 }}
                      className={`flex items-center gap-3 rounded-xl px-4 py-3.5 border-2 bg-white transition-colors
                        ${error ? 'border-red-400' : phone.length > 0 ? 'border-royal-500' : 'border-gray-200'}`}>
                      <span className="text-sm font-bold text-gray-400 shrink-0">+91</span>
                      <div className="w-px h-4 bg-gray-200 shrink-0" />
                      <span className={`flex-1 text-xl font-black tracking-[0.1em] ${error ? 'text-red-500' : 'text-gray-900'} ${phone.length === 0 ? 'opacity-0' : ''}`}>
                        {phone.length > 0 ? phone.replace(/(\d{5})(\d{0,5})/, '$1 $2').trim() : '0'}
                      </span>
                      {phone.length > 0 && <span className="text-[10px] text-gray-300 shrink-0 font-mono">{phone.length}/10</span>}
                    </motion.div>

                    <ErrorLine
                      show={!!(error || phoneCountdown > 0 || phoneLockUntil === -1)}
                      isLock={phoneLockUntil === -1 || phoneCountdown > 0}
                      text={phoneLockUntil === -1 ? 'Device blocked. Contact admin.' : phoneCountdown > 0 ? `Locked Aú ${fmtTime(phoneCountdown)}` : error}
                    />

                    {/* numpad - mobile only */}
                    <div className="lg:hidden">
                      <NumPad disabled={isPhoneLocked} onPress={(k) => {
                        setError('');
                        if (k === 'CLEAR') setPhone('');
                        else if (k === 'ƒO®') setPhone(p => p.slice(0, -1));
                        else if (phone.length < 10) setPhone(p => p + k);
                      }} />
                    </div>

                    <p className="hidden lg:block text-xs text-gray-300 mt-3">Type on keyboard Aú Enter to continue</p>

                    <motion.button whileTap={{ scale: 0.97 }} onClick={handlePhoneSubmit}
                      disabled={loading || phone.length < 10 || isPhoneLocked}
                      className={`w-full mt-5 font-bold py-3.5 rounded-xl flex items-center justify-center gap-2 text-sm transition-all
                        ${phone.length === 10 && !isPhoneLocked
                          ? 'bg-royal-700 text-white shadow-md shadow-royal-700/20 hover:bg-royal-600'
                          : 'bg-gray-100 text-gray-400 cursor-not-allowed'}`}>
                      {loading
                        ? <motion.div animate={{ rotate: 360 }} transition={{ repeat: Infinity, duration: 0.75 }} className="w-4 h-4 border-2 border-white/30 border-t-white rounded-full" />
                        : <>Continue <ArrowRight size={15} /></>}
                    </motion.button>
                  </motion.div>
                )}
